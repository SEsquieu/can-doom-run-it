find_package(Threads REQUIRED)
add_library(cdr_io STATIC control_bus.cpp udp_output.cpp keyboard_adapter.cpp keyboard_endpoint.cpp)
target_link_libraries(cdr_io PRIVATE Threads::Threads)
target_compile_features(cdr_io PRIVATE cxx_std_17)
if(WIN32)
    target_link_libraries(cdr_io PUBLIC ws2_32 user32)
else()
    find_path(CDR_XCB_INCLUDE_DIR xcb/xcb.h REQUIRED)
    find_library(CDR_XCB_LIBRARY NAMES xcb libxcb.so.1 REQUIRED)
    target_include_directories(cdr_io PRIVATE "${CDR_XCB_INCLUDE_DIR}")
    target_link_libraries(cdr_io PUBLIC "${CDR_XCB_LIBRARY}")
endif()
add_executable(doom-io-input-lab input_lab.cpp)
target_link_libraries(doom-io-input-lab PRIVATE Threads::Threads)
if(WIN32)
    target_link_libraries(doom-io-input-lab PRIVATE user32 gdi32)
else()
    target_include_directories(doom-io-input-lab PRIVATE "${CDR_XCB_INCLUDE_DIR}")
    target_link_libraries(doom-io-input-lab PRIVATE "${CDR_XCB_LIBRARY}")
endif()
set_target_properties(doom-io-input-lab PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
